cmake_minimum_required(VERSION 3.16)
project(limit_order_book VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

option(LOB_BUILD_TESTS   "Build the C++ test-suite"             ON)
option(LOB_BUILD_BENCH   "Build the benchmark"                  ON)
option(LOB_BUILD_APPS    "Build the lob_cli replay tool"        ON)
option(LOB_BUILD_PYTHON  "Build the pybind11 module in-tree"    OFF)
option(LOB_NATIVE        "Compile with -march=native"           OFF)
option(LOB_SANITIZE      "Enable AddressSanitizer + UBSan"      OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release)
endif()

add_library(lob_warnings INTERFACE)
if(MSVC)
  target_compile_options(lob_warnings INTERFACE /W4)
else()
  target_compile_options(lob_warnings INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
  if(LOB_SANITIZE)
    target_compile_options(lob_warnings INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer -g)
    target_link_options(lob_warnings INTERFACE -fsanitize=address,undefined)
  endif()
endif()

add_library(lob_core STATIC src/order_book.cpp src/types.cpp)
target_include_directories(lob_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_link_libraries(lob_core PUBLIC lob_warnings)
set_target_properties(lob_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(LOB_NATIVE AND NOT MSVC)
  target_compile_options(lob_core PUBLIC -march=native)
endif()
if(NOT MSVC)
  target_compile_options(lob_core PUBLIC $<$<CONFIG:Release>:-O3>)
endif()

if(LOB_BUILD_APPS)
  add_executable(lob_cli apps/lob_cli.cpp)
  target_link_libraries(lob_cli PRIVATE lob_core)
endif()

if(LOB_BUILD_BENCH)
  add_executable(lob_bench benchmarks/bench.cpp)
  target_link_libraries(lob_bench PRIVATE lob_core)
endif()

if(LOB_BUILD_TESTS)
  enable_testing()
  # NB: tests use their own CHECK macro, NOT assert(), so they still mean something under -DNDEBUG.
  add_executable(lob_tests tests/main.cpp tests/test_order_book.cpp tests/test_differential.cpp)
  target_include_directories(lob_tests PRIVATE tests)
  target_link_libraries(lob_tests PRIVATE lob_core)
  add_test(NAME lob_tests COMMAND lob_tests)
endif()

if(LOB_BUILD_PYTHON)
  find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
                  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_core src/bindings.cpp)
  target_link_libraries(_core PRIVATE lob_core)
  set_target_properties(_core PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/python/lob")
endif()
